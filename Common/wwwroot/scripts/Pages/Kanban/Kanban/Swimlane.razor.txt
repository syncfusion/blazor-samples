@code{
    private IReadOnlyCollection<KanbanDataModel> CardData = KanbanDataModel.GetMarketingCampaigns();
    private IReadOnlyCollection<KanbanDataModel> SortData = KanbanDataModel.SortDropDowns();
    private SortDirection _sortDirection { get; set; } = SortDirection.Ascending;
    private bool _allowDragAndDrop { get; set; } = false;
    private bool _showItemCount { get; set; } = true;
    private bool _enableFrozenRows { get; set; } = false;
    private int? _index { get; set; } = 0;
    
    private List<ColumnModel> ColumnData = new List<ColumnModel>() 
    {
        new ColumnModel(){ HeaderText= "Idea", KeyField= new List<string>() { "Idea" } },
        new ColumnModel(){ HeaderText= "Drafting", KeyField= new List<string>() { "Drafting"} },
        new ColumnModel(){ HeaderText= "Review", KeyField= new List<string>() { "Review" } },
        new ColumnModel(){ HeaderText= "Live", KeyField= new List<string>() { "Live" } }
    };

    private void ChangeSortOrder(Syncfusion.Blazor.DropDowns.ChangeEventArgs<string, KanbanDataModel> args)
    {
        this._sortDirection = (args.Value == "Ascending") ? SortDirection.Ascending : SortDirection.Descending;
    }
}
