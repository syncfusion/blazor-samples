<p>
    Key interactive features demonstrated in this sample include:
</p>
<ul>
    <li><b>Initial State</b>: The <strong>Testing</strong> column is collapsed by default using the <code>IsExpanded</code> property.</li>
    <li><b>Header Toggle</b>: Use the chevron in any column header to collapse or expand the column on demand.</li>
    <li><b>Auto Layout</b>: The remaining columns automatically stretch to occupy the freed space when a column is collapsed.</li>
    <li><b>Card Persistence</b>: Cards in collapsed columns remain in the data source and reappear when the column is expanded again.</li>
</ul>
<p>
    The <b>Column Configuration</b> highlights the building blocks of toggle behavior:
</p>
<ul>
    <li><b>AllowToggle</b>: Enables or disables the toggle button on the column header.</li>
    <li><b>IsExpanded</b>: Controls the initial expand or collapse state of the column.</li>
    <li><b>Auto Resize</b>: The board rebalances the visible columns when a column is toggled.</li>
</ul>

@code{
    private IReadOnlyCollection<KanbanDataModel> CardData = KanbanDataModel.GetCourseScheduling();
    private List<ColumnModel> ColumnData = new List<ColumnModel>()
    {
        new ColumnModel() { HeaderText = "Proposed", KeyField = new List<string>() { "Proposed" }, AllowToggle = true, IsExpanded = true },
        new ColumnModel() { HeaderText = "Approved", KeyField = new List<string>() { "Approved" }, AllowToggle = true, IsExpanded = true },
        new ColumnModel() { HeaderText = "In Session", KeyField = new List<string>() { "In Session" }, AllowToggle = true, IsExpanded = false },
        new ColumnModel() { HeaderText = "Archived", KeyField = new List<string>() { "Archived" }, AllowToggle = true, IsExpanded = true }
    };
        private string GetInitials(string? name)
    {
        if (string.IsNullOrEmpty(name)) return "?";
        var parts = name.Split(' ');
        return parts.Length > 1 ? $"{parts[0][0]}{parts[1][0]}".ToUpper() 
                                 : name[0].ToString().ToUpper();
    }
}
