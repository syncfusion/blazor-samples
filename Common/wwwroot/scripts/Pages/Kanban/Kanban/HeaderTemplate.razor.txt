<div class="col-lg-12 control-section">
    <SfKanban KeyField="Status" DataSource="@CardData" Height="500px">
        <KanbanColumns>
            @foreach (ColumnModel item in ColumnData)
            {
                <KanbanColumn HeaderText="@item.HeaderText" KeyField="@item.KeyField" AllowToggle="@item.AllowToggle">
                    <Template>
                        @{
                            var column = context as KanbanColumn;
                            string iconClass;
                            if (column.KeyField != null && column.KeyField.Count > 0)
                            {
                                iconClass = column.KeyField[0];
                            }
                            else
                            {
                                iconClass = string.Empty;
                            }
                        }
                        <div class="header-template-wrap">
                            <div class="header-icon e-icons @iconClass"></div>
                            <div class="header-text">@column.HeaderText</div>
                        </div>
                    </Template>
                </KanbanColumn>
            }
        </KanbanColumns>
        <KanbanCardSettings ContentField="Summary" HeaderField="Id"></KanbanCardSettings>
    </SfKanban>
</div>
@code{
    private IReadOnlyCollection<KanbanDataModel> CardData = KanbanDataModel.GetHotelHousekeeping();
    private List<ColumnModel> ColumnData = new List<ColumnModel>()
    {
        new ColumnModel() { HeaderText = "Dirty", KeyField = new List<string>() { "Dirty" } },
        new ColumnModel() { HeaderText = "In Progress", KeyField = new List<string>() { "In Progress" } },
        new ColumnModel() { HeaderText = "Cleaned", KeyField = new List<string>() { "Cleaned" } },
        new ColumnModel() { HeaderText = "Inspected", KeyField = new List<string>() { "Inspected" } }
    };
}
