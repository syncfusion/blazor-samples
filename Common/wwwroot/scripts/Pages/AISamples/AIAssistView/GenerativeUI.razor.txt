    private SfAIAssistView sfAIAssistView = new();
    private List<string> suggestions = AssistviewData.GetGenerativeSuggestionData().ToList();

    private async Task PromptRequest( AssistViewPromptRequestedEventArgs args)
    {
        try
        {
            var json = await AzureAIService.GetCompletionAsync(
                    args.Prompt,
                    returnAsJson: false,
                    appendPreviousResponse: false,
                    systemRole: AssistviewData.GetGenerativeUISystemPrompt());

            var aiResponse =
                JsonSerializer.Deserialize<AiResponse>(
                    json,
                    new JsonSerializerOptions
                    {
                        PropertyNameCaseInsensitive = true
                    });

            if (aiResponse == null)
            {
                await sfAIAssistView.UpdateResponseAsync(
                    "Unable to parse AI response.");
                return;
            }

            List<IResponseBlock> responseBlocks = new();

            foreach (var block in aiResponse.Blocks)
            {
                switch (block.BlockType.ToLower())
                {
                    case "text":

                        responseBlocks.Add(new TextBlock
                        {
                            BlockType = BlockType.Text,
                            Content = block.Content
                        });

                        break;

                    case "thinking":

                        responseBlocks.Add(new ThinkingBlock
                        {
                            BlockType = BlockType.Thinking,
                            Title = string.IsNullOrWhiteSpace(block.Title)
                                ? "Reasoning"
                                : block.Title,
                            Content = block.Content,
                            Collapsible = true,
                            Collapsed = true
                        });

                        break;

                    case "tool":

                        object? props = null;

                        if (block.ToolName == "weather-card")
                        {
                            props = JsonSerializer.Deserialize<WeatherProps>( block.Props.GetRawText(),
                                    new JsonSerializerOptions
                                    {
                                        PropertyNameCaseInsensitive = true
                                    });
                        }
                        else if (block.ToolName == "grid-tool")
                        {
                            var gridProps = JsonSerializer.Deserialize<GridToolProps>( block.Props.GetRawText(),
                                    new JsonSerializerOptions
                                    {
                                        PropertyNameCaseInsensitive = true
                                    });

                            props = new
                            {
                                gridData = ParseGridData(gridProps),
                                columns = gridProps?.Columns
                            };
                        }
                        else if (block.ToolName == "chart-tool")
                        {
                            var chartProps = JsonSerializer.Deserialize<ChartToolProps>( block.Props.GetRawText(),
                                    new JsonSerializerOptions
                                    {
                                        PropertyNameCaseInsensitive = true
                                    });

                            var chartData = ParseChartData(
                                chartProps,
                                chartProps.XField,
                                chartProps.YField);

                            props = new
                            {
                                chartData,
                                xField = chartProps.XField,
                                yField = chartProps.YField,
                                chartType = chartProps.ChartType,
                                title = chartProps.Title,
                                xAxisTitle = chartProps.XAxisTitle,
                                yAxisTitle = chartProps.YAxisTitle,
                                enableTooltip = chartProps.EnableTooltip ?? true,
                                enableLegend = chartProps.EnableLegend ?? true
                            };

                        }

                        responseBlocks.Add(new ToolBlock
                        {
                            BlockType = BlockType.Tool,
                            ToolName = block.ToolName,
                            Props = props
                        });

                        break;
                }
            }

            await sfAIAssistView.UpdateResponseAsync(string.Empty, responseBlocks);
        }
        catch (Exception ex)
        {
            await sfAIAssistView.UpdateResponseAsync("We could not reach the AI service; please try again later.");
        }
    }

    private void ToolbarItemClicked(AssistViewToolbarItemClickedEventArgs args)
    {
        if (args.Item.IconCss == "e-icons e-refresh")
        {
            sfAIAssistView.Prompts.Clear();
        }
    }

    private static ChartSeriesType GetChartSeriesType(string chartType)
    {
        return chartType?.ToLower() switch
        {
            "column" => ChartSeriesType.Column,
            "bar" => ChartSeriesType.Bar,
            "area" => ChartSeriesType.Area,
            "spline" => ChartSeriesType.Spline,
            "steparea" => ChartSeriesType.StepArea,
            "step" => ChartSeriesType.StepLine,
            "scatter" => ChartSeriesType.Scatter,
            _ => ChartSeriesType.Line,
        };
    }

    private static IEnumerable<ChartDataPoint> ParseChartData(ChartToolProps chartProps, string xField, string yField)
    {
        var element = chartProps.DataSource.ValueKind != JsonValueKind.Undefined && chartProps.DataSource.ValueKind != JsonValueKind.Null
            ? chartProps.DataSource
            : chartProps.Data.ValueKind != JsonValueKind.Undefined && chartProps.Data.ValueKind != JsonValueKind.Null
                ? chartProps.Data
                : default;

        if (element.ValueKind != JsonValueKind.Array)
        {
            return Array.Empty<ChartDataPoint>();
        }

        var result = new List<ChartDataPoint>();
        foreach (var item in element.EnumerateArray())
        {
            if (item.ValueKind != JsonValueKind.Object)
            {
                continue;
            }

            var xValue = item.TryGetProperty(xField, out var xProp)
                ? xProp.ToString()
                : string.Empty;

            var yValue = item.TryGetProperty(yField, out var yProp)
                ? yProp
                : default;

            var yDouble = 0.0;
            if (yValue.ValueKind == JsonValueKind.Number)
            {
                yDouble = yValue.GetDouble();
            }
            else if (yValue.ValueKind == JsonValueKind.String &&
                     double.TryParse(yValue.GetString(), out var parsed))
            {
                yDouble = parsed;
            }

            result.Add(new ChartDataPoint
            {
                X = xValue ?? string.Empty,
                Y = yDouble
            });
        }

        return result;
    }

    private static IEnumerable<ExpandoObject> ParseGridData(GridToolProps gridProps)
    {
        var element = gridProps.GridData.ValueKind != JsonValueKind.Undefined && gridProps.GridData.ValueKind != JsonValueKind.Null
            ? gridProps.GridData
            : gridProps.Data.ValueKind != JsonValueKind.Undefined && gridProps.Data.ValueKind != JsonValueKind.Null
                ? gridProps.Data
                : default;

        if (element.ValueKind != JsonValueKind.Array)
        {
            return Array.Empty<ExpandoObject>();
        }

        var rows = new List<ExpandoObject>();

        foreach (var item in element.EnumerateArray())
        {
            if (item.ValueKind != JsonValueKind.Object)
            {
                continue;
            }

            var row = new ExpandoObject();
            var dict = (IDictionary<string, object>)row;

            foreach (var property in item.EnumerateObject())
            {
                dict[property.Name] = property.Value.ValueKind switch
                {
                    JsonValueKind.String => property.Value.GetString() ?? string.Empty,
                    JsonValueKind.Number => property.Value.TryGetInt64(out var l)
                        ? l
                        : property.Value.GetDouble(),
                    JsonValueKind.True => true,
                    JsonValueKind.False => false,
                    JsonValueKind.Null => string.Empty,
                    _ => property.Value.ToString() ?? string.Empty,
                };
            }

            rows.Add(row);
        }

        return rows;
    }
