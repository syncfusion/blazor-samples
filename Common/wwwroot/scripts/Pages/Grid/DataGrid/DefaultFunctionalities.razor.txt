    /// <summary>Collection of transaction records bound to the grid via OnDataLoad.</summary>
    private List<TransactionRecord>? GridData { get; set; }
    /// <summary>Initialize grid with transaction records when component loads.</summary>
    protected override void OnInitialized()
    {
        // Load predefined transaction records from data source
        GridData = TransactionData.GetAllRecords().ToList();
    }

    /// <summary>Handle data load requests from the grid and apply sort/paging operations.</summary>
    private async Task HandleDataLoad(GridLoadEventArgs<TransactionRecord> args)
    {
        var result = ApplyDataOperations(GridData!,args.Request.Sort!,args.Request.Skip,args.Request.Take);
        args.Result = result.Result;
        args.Count = result.Count;
        await Task.CompletedTask;
    }

    /// <summary>
    /// Apply sort, paging, and other operations to the data source efficiently.
    /// </summary>
    private static (IEnumerable<TransactionRecord> Result, int Count) ApplyDataOperations(IEnumerable<TransactionRecord> source, List<SortRequest> sorted, int skip,int take)
    {
        var sort = sorted?.FirstOrDefault();
        if (sort != null)
        {
            var property = typeof(TransactionRecord).GetProperty(sort.Name!);
            if (property != null)
            {
                source = sort.Direction == "ascending" ? source.OrderBy(x => property.GetValue(x)) : source.OrderByDescending(x => property.GetValue(x));
            }
        }
        var count = source.Count();
        if (skip != 0 || take != 0)
        {
            source = source.Skip(skip).Take(take);
        }
        return (Result: source, Count: count);
    }

}
