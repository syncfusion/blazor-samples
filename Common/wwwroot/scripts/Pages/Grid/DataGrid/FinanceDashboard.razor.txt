    private void OpenDetailsDialog(StockItem stock)
    {
        SelectedStock = stock;
        DialogHeader = $"{stock.Ticker} — {stock.CompanyName}";
        ChartBorderColor = "#0288d1";

        // Build chart points from the live trend data
        DialogChartData = stock.TrendData
            .Select((p, i) => new DialogChartPoint { Index = i, Price = (double)p })
            .ToList();

        IsDetailsDialogVisible = true;
    }

    private void OnDialogClosed(CloseEventArgs args)
    {
        SelectedStock = null;
        DialogChartData = new();
        IsInitialRender = false;
    }
    private void OnDialogOpen(Object args)
    {
        IsInitialRender = true;
    }
    private static string FormatMarketCap(decimal value)
    {
        double v = (double)value;
        if (v >= 1_000_000_000_000) return (v / 1_000_000_000_000).ToString("F2") + "T";
        if (v >= 1_000_000_000) return (v / 1_000_000_000).ToString("F2") + "B";
        if (v >= 1_000_000) return (v / 1_000_000).ToString("F2") + "M";
        if (v >= 1_000) return (v / 1_000).ToString("F2") + "K";
        return v.ToString("F2");
    }

            await Task.Delay(2000);
            IsLoading = false;
            await InvokeAsync(StateHasChanged);
            await StartTimer();
        }
    }

    private async Task StartTimer()
    {
        _timer = new PeriodicTimer(TimeSpan.FromMilliseconds(1000));
        while (await _timer.WaitForNextTickAsync())
        {
            await TickAsync();
        }
    }

    private readonly byte[] _randomBuffer = new byte[8];
    private async Task TickAsync()
    {
        foreach (var stock in StockData)
        {
            DataSource.FillRandomBytes(_randomBuffer);
            var randomValue = BitConverter.ToUInt64(_randomBuffer);
            var deltaPct = ((randomValue / (double)ulong.MaxValue) - 0.495) * 0.025;
            var prevPrice = stock.LastPrice;
            var newPrice = Math.Round(Math.Max(prevPrice * (decimal)(1 + deltaPct), 0.01m) * 100) / 100;
            var change = Math.Round((newPrice - prevPrice) * 100) / 100;
            var changePct = Math.Round((change / prevPrice) * 100 * 1000) / 1000;
            var vol = DataSource.GenerateVolume();
            var newMktCap = (long)Math.Round(stock.MarketCap * (1 + (decimal)deltaPct * 0.1m));
            var isGain = change >= 0;

            // Update stock data
            stock.LastPrice = newPrice;
            stock.Change = change;
            stock.ChangePercent = changePct;
            stock.IsGain = isGain;
            stock.Volume = vol;
            stock.MarketCap = newMktCap;

            // Update high and low
            if (newPrice > stock.High)
                stock.High = newPrice;
            if (newPrice < stock.Low)
                stock.Low = newPrice;

            // Update trend data
            if (stock.TrendData.Count >= 20)
                stock.TrendData.RemoveAt(0);
            stock.TrendData.Add(newPrice);
            stock.SparkLinePoints.Clear();
            Collection<SparklinePoint> sparkPoints = new Collection<SparklinePoint>();
            foreach (var point in stock.TrendData.Select((p, i) => new SparklinePoint { Index = i, Price = (double)p }).ToList())
            {
                sparkPoints.Add(point);
            }
            stock.SparkLinePoints = sparkPoints;
        }

        // Refresh dialog chart live if it's open and showing the same stock
        if (IsDetailsDialogVisible && SelectedStock != null)
        {
            DialogChartData = SelectedStock.TrendData
                .Select((p, i) => new DialogChartPoint { Index = i, Price = (double)p })
                .ToList();
            ChartBorderColor = "#0288d1";
        }

        // Re-apply sort to maintain sort state during real-time updates
        if (_isUserSorted)
        {
            ApplySort();

            if (_isUserSorting == false)
            {
                _isUserSorted = false;
            }
        }

        await InvokeAsync(StateHasChanged);
    }

    private void ApplySort()
    {
        if (string.IsNullOrEmpty(CurrentSortField))
            return;

        StockData = CurrentSortDirection == SortDirection.Ascending
            ? StockData.OrderBy(s => GetSortValue(s, CurrentSortField)).ToList()
            : StockData.OrderByDescending(s => GetSortValue(s, CurrentSortField)).ToList();
    }

    private object? GetSortValue(StockItem stock, string field)
    {
        return field switch
        {
            "Ticker" => stock.Ticker,
            "CompanyName" => stock.CompanyName,
            "Sector" => stock.Sector,
            "LastPrice" => stock.LastPrice,
            "Change" => stock.Change,
            "ChangePercent" => stock.ChangePercent,
            "High" => stock.High,
            "Low" => stock.Low,
            "Volume" => stock.Volume,
            "MarketCap" => stock.MarketCap,
            _ => stock.Id
        };
    }
    private bool _isUserSorted { get; set; } = false;
    private bool _isUserSorting { get; set; } = false;
    private async Task OnGridActionComplete(ActionEventArgs<StockItem> args)
    {
        // Track sort changes when user clicks column headers
        if (args.RequestType == Syncfusion.Blazor.Grids.Action.Sorting)
        {
            if (GridRef?.CurrentViewData?.Count() > 0)
            {
                var sortSettings = GridRef.SortSettings;
                if (sortSettings?.Columns?.Count > 0)
                {
                    CurrentSortField = sortSettings.Columns[0].Field;
                    CurrentSortDirection = sortSettings.Columns[0].Direction;
                    _isUserSorting = true;
                    _isUserSorted = true;
                }
                else
                {
                    CurrentSortField = "Id";
                    CurrentSortDirection = SortDirection.Ascending;
                    _isUserSorting = false;
                }
            }
        }
    }

    async ValueTask IAsyncDisposable.DisposeAsync()
    {
        _timer?.Dispose();
        await Task.CompletedTask;
    }
