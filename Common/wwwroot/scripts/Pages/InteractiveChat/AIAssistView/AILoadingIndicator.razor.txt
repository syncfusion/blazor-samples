<ActionDescription>
    <p>
        This sample demonstrates the loading indicator styles that can be displayed while the AI AssistView is generating a response.
    </p>

    <p>
        In this example, a custom loading indicator can be rendered while the AI response is being generated. The selected loading type is shown before the actual response is rendered using the <code>ResponseAnimationTemplate</code> property and removed once the response is added.
    </p>
    <p><b>Key Features:</b></p>
    <ul>
        <li>
            <strong>Dot type:</strong> Three continuously blinking dots with a smooth scale and opacity animation.
        </li>
        <li>
            <strong>Spinner type:</strong> A single circular spinner rotating continuously until the response is fetched.
        </li>
        <li>
            <strong>Text type:</strong> Status messages that cycle through three stages: "🧠 Understanding request..." → "✍️ Drafting response..." → "🚀 Almost ready...".
        </li>
        <li>
            <strong>Text with indicator type:</strong> "Generating" text with three animated dots as a waiting indicator until the AI response is received.
        </li>
    </ul>

    <p>
        Switch between the loading types in the property pane to preview each loading animation style while responses are being generated.
    </p>

</ActionDescription>

@code {
    private SfAIAssistView AIAssist = new SfAIAssistView();
    private string selectedLoadingType = "dot";
    private List<string> suggestions = PromptResponseData.GetAllSuggestionData().ToList();
    private IReadOnlyList<PromptResponseData> prompts = PromptResponseData.GetAllPromptResponseData();
    private List<LoadingTypeItem> loadingTypes = new List<LoadingTypeItem>()
    {
        new LoadingTypeItem() { Text = "Dot", Value = "dot" },
        new LoadingTypeItem() { Text = "Spinner", Value = "spinner" },
        new LoadingTypeItem() { Text = "Text", Value = "text" },
        new LoadingTypeItem() { Text = "Text with indicator", Value = "textIndicator" }
    };
    private void OnLoadingTypeChanged(ChangeEventArgs<string, LoadingTypeItem> args)
    {
        selectedLoadingType = args.Value;
    }
    private async Task PromptRequest(AssistViewPromptRequestedEventArgs args)
    {
        await Task.Delay(3000);
        var promptData = prompts.FirstOrDefault(prompt => prompt.Prompt == args.Prompt);
        var defaultResponse = "For real-time prompt processing, connect the AI AssistView control to your preferred AI service, such as OpenAI or Azure Cognitive Services. Ensure you obtain the necessary API credentials to authenticate and enable seamless integration.";
        args.Response = promptData?.Response ?? defaultResponse;
        args.PromptSuggestions = promptData?.SuggestionData.Count > 0 ? promptData.SuggestionData.ToList() : suggestions;
    }
    private void ToolbarItemClicked(AssistViewToolbarItemClickedEventArgs args)
    {
        if (args.Item.IconCss == "e-icons e-refresh")
        {
            AIAssist.Prompts.Clear();
        }
    }
    public class LoadingTypeItem
    {
        public string Text { get; set; } = string.Empty;
        public string Value { get; set; } = string.Empty;
    }
}
