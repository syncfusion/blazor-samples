    private Theme Theme { get; set; }
    public string Width { get; set; } = "90%";
    public string Height { get; set; } = "600px";
    public double Radius { get; set; } = 0.9;
    public string FontFamily { get; set; } = "Roboto";

    private SfSunburstChart<NodeDetails>? ChartRef { get; set; }

    private string[] Palette = Array.Empty<string>();

    private string FileName { get; set; } = "SunburstChart";
    private ExportType SelectedExportType { get; set; } = ExportType.JPEG;

    public List<ExportFormat> ExportFormats { get; set; } = new List<ExportFormat>
    {
        new ExportFormat() { Name = "JPEG", Type = ExportType.JPEG },
        new ExportFormat() { Name = "PNG", Type = ExportType.PNG },
        new ExportFormat() { Name = "SVG", Type = ExportType.SVG },
        new ExportFormat() { Name = "PDF", Type = ExportType.PDF },
        new ExportFormat() { Name = "XLSX", Type = ExportType.XLSX },
        new ExportFormat() { Name = "CSV", Type = ExportType.CSV }
    };

    public class ExportFormat
    {
        public string Name { get; set; } = string.Empty;
        public ExportType Type { get; set; }
    }

    protected override void OnInitialized()
    {
        Theme = ThemeHelper.GetCurrentTheme(NavigationManager.Uri);
        FontFamily = ThemeHelper.GetFontFamily(NavigationManager.Uri);
        Palette = SunburstThemeColors.GetPrintExportPalette(Theme);

        @*Hidden:Lines*@
        if (SampleService != null && SampleService.IsDevice)
        {
            Width = "100%";
            Height = "360px";
            Radius = 0.85;
        }
        @*End:Hidden*@
    }

    public class NodeDetails
    {
        public string Id { get; set; } = string.Empty;
        public string ParentId { get; set; } = string.Empty;
        public string Name { get; set; } = string.Empty;
        public double Value { get; set; }
    }

    public async Task PrintChart(MouseEventArgs args)
    {
        if (ChartRef != null)
        {
            await ChartRef.PrintAsync();
        }
    }

    public async Task ExportChart(MouseEventArgs args)
    {
        if (ChartRef == null)
        {
            return;
        }

        var format = ExportFormats.FirstOrDefault(f => f.Type == SelectedExportType)
                     ?? ExportFormats.FirstOrDefault()
                     ?? new ExportFormat { Type = ExportType.PNG };

        await ChartRef.ExportAsync(format.Type, string.IsNullOrWhiteSpace(FileName) ? "SunburstChart" : FileName);
    }
