<ActionDescription>
<p>This Blazor Sunburst Chart example visualizes <b>Customer Effort Score Distribution (in Percent of Customer Journeys)</b> across touchpoint tiers (First, Second, Third, Fourth, Fifth) categorized by engagement channels and interaction types.</p>

<p>In this example, you can see how to configure selection and highlight behaviors in a sunburst chart with an asymmetric multi-tier radial hierarchy. By default, both selection and highlight are disabled. Use the property panel to enable selection or highlight, then choose the desired mode to explore hierarchical relationships. Key features demonstrated here include:</p>
<ul>
    <li><code>SunburstSelectionSettings</code> with <code>Enable</code> toggleable from the property panel to allow clicking and selecting a segment</li>
    <li><code>Mode</code> options such as <code>Child</code>, <code>Parent</code>, <code>Single</code>, and <code>All</code> to control selection scope</li>
    <li><code>SunburstHighlightSettings</code> with <code>Enable</code> toggleable to visually emphasize a segment on hover</li>
    <li>Asymmetric depth levels ranging from 2 up to 4 rings producing an uneven sunburst flare silhouette</li>
    <li><code>SunburstDataLabelSettings</code> with <code>LabelRotationMode</code> set to <code>Angle</code> and <code>LabelOverflowMode</code> set to <code>Trim</code> for readable curved labels</li>
    <li>A custom <code>Palette</code> aligned to the active theme to color the rings consistently</li>
</ul>

<p><code>Tooltip</code> is enabled in this example. To see tooltip details, hover over or tap on a segment. Enable selection and highlight from the property panel to interact with the chart.</p>

<p>More information about the sunburst chart selection and highlight can be found in this <a target='_blank' href='https://blazor.syncfusion.com/documentation/sunburst-chart/selection-highlight' aria-label="Navigate to the documentation for sunburst chart selection and highlight" rel='noopener'>documentation section</a>.</p>
</ActionDescription>

    private Theme Theme { get; set; }
    public string Width { get; set; } = "90%";
    public string Height { get; set; } = "600px";
    public double Radius { get; set; } = 0.9;
    public string FontFamily { get; set; } = "Roboto";

    private SfSunburstChart<NodeDetails>? ChartRef { get; set; }

    private string[] Palette = Array.Empty<string>();

    private bool IsSelectionEnabled = false;
    private bool IsHighlightEnabled = false;
    private SunburstSelectionMode SelectionModeValue = SunburstSelectionMode.Child;
    private SunburstHighlightMode HighlightModeValue = SunburstHighlightMode.Parent;

    public List<DropDownData> SelectionModes { get; set; } = new List<DropDownData>
    {
        new DropDownData() { Type = "Child" },
        new DropDownData() { Type = "Parent" },
        new DropDownData() { Type = "Single" },
        new DropDownData() { Type = "All" }
    };

    public List<DropDownData> HighlightModes { get; set; } = new List<DropDownData>
    {
        new DropDownData() { Type = "Child" },
        new DropDownData() { Type = "Parent" },
        new DropDownData() { Type = "Single" },
        new DropDownData() { Type = "All" }
    };

    protected override void OnInitialized()
    {
        Theme = ThemeHelper.GetCurrentTheme(NavigationManager.Uri);
        FontFamily = ThemeHelper.GetFontFamily(NavigationManager.Uri);
        Palette = SunburstThemeColors.GetSelectionPalette(Theme);

        @*Hidden:Lines*@
        if (SampleService != null && SampleService.IsDevice)
        {
            Width = "100%";
            Height = "360px";
            Radius = 0.85;
        }
        @*End:Hidden*@
    }

    public class NodeDetails
    {
        public string Id { get; set; } = string.Empty;
        public string ParentId { get; set; } = string.Empty;
        public string Name { get; set; } = string.Empty;
        public double Value { get; set; }
    }

    public class DropDownData
    {
        public string? Type { get; set; }
    }

    private async Task ChangeSelection(Syncfusion.Blazor.Buttons.ChangeEventArgs<bool> args)
    {
        await Task.CompletedTask;
        IsSelectionEnabled = args.Checked;
    }

    private async Task ChangeSelectionMode(Syncfusion.Blazor.DropDowns.ChangeEventArgs<string, DropDownData> args)
    {
        await Task.CompletedTask;
        SelectionModeValue = (SunburstSelectionMode)Enum.Parse(typeof(SunburstSelectionMode), args?.Value?.ToString() ?? string.Empty, true);
    }

    private async Task ChangeHighlight(Syncfusion.Blazor.Buttons.ChangeEventArgs<bool> args)
    {
        await Task.CompletedTask;
        IsHighlightEnabled = args.Checked;
    }

    private async Task ChangeHighlightMode(Syncfusion.Blazor.DropDowns.ChangeEventArgs<string, DropDownData> args)
    {
        await Task.CompletedTask;
        HighlightModeValue = (SunburstHighlightMode)Enum.Parse(typeof(SunburstHighlightMode), args?.Value?.ToString() ?? string.Empty, true);
    }
