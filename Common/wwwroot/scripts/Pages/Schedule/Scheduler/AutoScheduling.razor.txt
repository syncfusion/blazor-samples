<p>

    private bool IsTimeSlotAvailableForResource(int resourceId, DateTime startTime, DateTime endTime)
    {
        if (ScheduleRef == null)
            return false;

        var allEvents = ScheduleDatasource;
        var proposedDay = new DateTime(startTime.Year, startTime.Month, startTime.Day);

        return !allEvents.Any(e =>
        {
            if (e.StaffId != resourceId)
                return false;

            var eventDay = new DateTime(e.StartTime.Year, e.StartTime.Month, e.StartTime.Day);
            if (eventDay != proposedDay)
                return false;

            return startTime < e.EndTime && endTime > e.StartTime;
        });
    }

    public async Task ScheduleDragStop(DragEventArgs<AppointmentData> args)
    {
        var gridData = await JSRuntime.InvokeAsync<bool>("isSelectorFromPoint", args.MouseEventArgs.ClientX, args.MouseEventArgs.ClientY, "grid-content");
        if (gridData)
        {
            var schedule = ScheduleRef;
            var grid = GridRef;
            if (schedule == null || grid == null)
                return;

            args.Cancel = true;
            double duration = (args.EndTime - args.StartTime).TotalHours;
            AppointmentData data = new AppointmentData
            {
                Id = args.Data.Id,
                StartTime = args.StartTime,
                EndTime = args.EndTime,
                Subject = args.Data.Subject,
                StaffId = args.Data.StaffId,
                RequiredSkill = args.Data.RequiredSkill,
                IsAllDay = args.Data.IsAllDay
            };
            ScheduleDatasource.Remove(args.Data);
            await ScheduleRef!.RefreshEventsAsync();
        }
        if (DraggedEvent != null)
        {
            var resource = ResourceDatasource
                .FirstOrDefault(r => r.Id == args.Data.StaffId);
        
            if (resource != null &&
                !resource.Skills!.Contains(args.Data.RequiredSkill ?? ""))
            {
                args.Cancel = true;
                return;
            }

            double durationHours = (args.Data.EndTime - args.Data.StartTime).TotalHours;
            double currentWorkload = GetResourceWorkloadForDate(args.Data.StaffId,args.Data.StartTime,args.Data.Id);
            if (currentWorkload + durationHours > MaxDailyWorkload)
            {
                args.Cancel = true;
                return;
            }
        }
        await InvokeAsync(StateHasChanged);
    }

    public async Task GridDragStop(RowDroppingEventArgs<GridAppointmentData> args)
    {
        if (args.TargetDimension != null)

                return;

            await Task.Yield();
            CellClickEventArgs? cellData = await schedule.GetTargetCellAsync((int)args.ClientX, (int)args.ClientY);

                    if (resource != null && resource.Skills != null && resource.Skills.Contains(appt.RequiredSkill ?? ""))
                    {
                        AppointmentData newEvent = new AppointmentData
                        {
                            Id = ScheduleDatasource.Max(e => e.Id) + 1,
                            Subject = appt.Task,
                            StartTime = cellData.StartTime,
                            EndTime = cellData.StartTime.AddHours(durationHours),
                            IsAllDay = cellData.IsAllDay,
                            StaffId = staffId,
                            RequiredSkill = appt.RequiredSkill
                        };
                        double currentWorkload = GetResourceWorkloadForDate(staffId, newEvent.StartTime, 0);

    private string GetResourceWorkload(int resourceId, DateTime date)
    {
        var dayStart = new DateTime(date.Year, date.Month, date.Day);
        var dayEnd = new DateTime(date.Year, date.Month, date.Day, 23, 59, 59);

        var allEvents = ScheduleDatasource;
        double totalHours = 0;

        foreach (var e in allEvents)
        {
            var eventDate = new DateTime(e.StartTime.Year, e.StartTime.Month, e.StartTime.Day);
            if (eventDate != dayStart)
                continue;

            if (e.StaffId == resourceId)
            {
                totalHours += (e.EndTime - e.StartTime).TotalHours;
            }
        }

        return $"{totalHours:F1}h";
    }

    private (DateTime startTime, DateTime endTime)? FindAvailableTimeSlotForResource(int resourceId, int durationHours, DateTime date)
    {
        var dayStart = new DateTime(date.Year, date.Month, date.Day);
        dayStart = new DateTime(dayStart.Year, dayStart.Month, dayStart.Day, 9, 0, 0);

        var dayEnd = new DateTime(date.Year, date.Month, date.Day, 23, 0, 0);

        var allEvents = ScheduleDatasource;
        var selectedDay = date.Date;

        var resourceEvents = allEvents
            .Where(e =>
                e.StaffId == resourceId &&
                e.StartTime.Date == selectedDay)
            .OrderBy(e => e.StartTime)
            .ToList();

        var durationMs = durationHours * 60 * 60 * 1000;

        if (resourceEvents.Count == 0)
        {
            if ((dayEnd - dayStart).TotalHours >= durationHours)
            {
                return (dayStart, dayStart.AddHours(durationHours));
            }
        }
        else
        {
            if ((resourceEvents[0].StartTime - dayStart).TotalHours >= durationHours)
            {
                return (dayStart, dayStart.AddHours(durationHours));
            }

            for (int i = 0; i < resourceEvents.Count - 1; i++)
            {
                var gapStart = resourceEvents[i].EndTime;
                var gapEnd = resourceEvents[i + 1].StartTime;
                if ((gapEnd - gapStart).TotalHours >= durationHours)
                {
                    return (gapStart, gapStart.AddHours(durationHours));
                }
            }

            var lastEventEnd = resourceEvents.Last().EndTime;
            if ((dayEnd - lastEventEnd).TotalHours >= durationHours)
            {
                return (lastEventEnd, lastEventEnd.AddHours(durationHours));
            }
        }

        return null;
    }

    private async Task HandleAutoScheduling()
    {
        if (ScheduleRef == null || GridRef == null)
            return;

        ResourceWorkload.Clear();
        ScheduledAppointments.Clear();

        var scheduleDate = CurrentDate;
        var dayStart = CurrentDate.Date;

        foreach (var resource in ResourceDatasource)
        {
            ResourceWorkload[resource.Id] = GetResourceWorkloadForDate(resource.Id,CurrentDate,0);
        }

        var appointmentsToSchedule = GridDatasource.ToList();
        int nextEventId = ScheduleDatasource.Max(e => e.Id) + 1;

        foreach (var appt in appointmentsToSchedule)
        {
            var durationStr = appt.Duration ?? "1 Hour";
            int durationHours = int.Parse(durationStr.Split(' ')[0]);
            var requiredSkill = appt.RequiredSkill ?? "General";

            var matchingResources = ResourceDatasource
                .Where(r => r.Skills != null && r.Skills.Contains(requiredSkill))
                .OrderBy(r => ResourceWorkload.ContainsKey(r.Id) ? ResourceWorkload[r.Id] : 0)
                .ToList();

            foreach (var resource in matchingResources)
            {
                var currentWorkload = ResourceWorkload.ContainsKey(resource.Id) ? ResourceWorkload[resource.Id] : 0;

                if (currentWorkload + durationHours <= MaxDailyWorkload)
                {
                    var timeSlot = FindAvailableTimeSlotForResource(resource.Id, durationHours, CurrentDate.Date);

                    if (timeSlot.HasValue)
                    {
                        var newEvent = new AppointmentData
                        {
                            Id = nextEventId++,
                            Subject = appt.Task,
                            StartTime = timeSlot.Value.startTime,
                            EndTime = timeSlot.Value.endTime,
                            IsAllDay = false,
                            StaffId = resource.Id,
                            RequiredSkill = requiredSkill
                        };

                        ScheduleDatasource.Add(newEvent);
                        ResourceWorkload[resource.Id] = currentWorkload + durationHours;
                        ScheduledAppointments.Add(appt.Task ?? "");
                        break;
                    }
                }
            }
        }

        var remainingAppointments = GridDatasource
            .Where(a => !ScheduledAppointments.Contains(a.Task ?? ""))
            .ToList();
        GridDatasource = remainingAppointments;

        await ScheduleRef.RefreshEventsAsync(); 
        StateHasChanged();
    }
    
    private double GetResourceWorkloadForDate(int resourceId, DateTime date, int excludeEventId)
    {
        var dayStart = new DateTime(date.Year, date.Month, date.Day);
        var allEvents = ScheduleDatasource;
        double totalHours = 0;

        foreach (var e in allEvents)
        {
            var eventDate = new DateTime(e.StartTime.Year, e.StartTime.Month, e.StartTime.Day);
            if (eventDate != dayStart)
                continue;

            if (e.StaffId == resourceId && e.Id != excludeEventId)
            {
                totalHours += (e.EndTime - e.StartTime).TotalHours;
            }
        }

        return totalHours;
    }
