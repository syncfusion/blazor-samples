    <p>
        This sample demonstrates how to generate a GS1 Data Matrix barcode using GS1-compliant data. Combining the high-density encoding capabilities of a Data Matrix symbol with GS1 Application Identifiers (AIs), this two-dimensional barcode enables compact and accurate storage of product and traceability information, making it ideal for healthcare, pharmaceutical, manufacturing, and supply chain applications.

    public void matrixEncodingChanged(ChangeEventArgs<string, MatrixEncoding> args)
    {
        encoding = args.Value!;
        CssClass = "e-success";
        switch (encoding)
        {
            case "Auto":
                dataMatrixEncoding = DataMatrixEncoding.Auto;
                break;
            case "ASCII":
                dataMatrixEncoding = DataMatrixEncoding.ASCII;
                break;
            case "ASCIINumeric":
                if (Regex.IsMatch(dataMatrixGenerator!.Value, @"^\d+$"))
                {
                    dataMatrixEncoding = DataMatrixEncoding.ASCIINumeric;
                }
                else
                {
                    CssClass = "e-error";
                }
                break;
            case "Base256":
                dataMatrixEncoding = DataMatrixEncoding.Base256;
                break;
